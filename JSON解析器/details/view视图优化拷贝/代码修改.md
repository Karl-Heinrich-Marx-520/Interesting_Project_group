class Lexer {
private:
    std::string_view m_input;   // 改成视图，零拷贝
    size_t m_pos;
    size_t m_line;
    size_t m_column;

public:
    // 单构造通吃所有：string/字面量/临时串，全部零拷贝
    Lexer(std::string_view input)
        : m_input(input), m_pos(0), m_line(1), m_column(1) {}

    std::vector<Token> tokenize() {
        std::vector<Token> tokens;
        while (!is_at_end()) {
            skip_whitespace_and_comments();
            if (is_at_end()) break;

            char c = peek();
            size_t current_line = m_line;
            size_t current_column = m_column;

            switch (c) {
                case '{': advance(); tokens.emplace_back(TokenType::LeftBrace, "{", current_line, current_column); break;
                case '}': advance(); tokens.emplace_back(TokenType::RightBrace, "}", current_line, current_column); break;
                case '[': advance(); tokens.emplace_back(TokenType::LeftBracket, "[", current_line, current_column); break;
                case ']': advance(); tokens.emplace_back(TokenType::RightBracket, "]", current_line, current_column); break;
                case ':': advance(); tokens.emplace_back(TokenType::Colon, ":", current_line, current_column); break;
                case ',': advance(); tokens.emplace_back(TokenType::Comma, ",", current_line, current_column); break;
                case '"': tokens.push_back(read_string()); break;
                case 't': tokens.push_back(read_keyword("true", TokenType::True)); break;
                case 'f': tokens.push_back(read_keyword("false", TokenType::False)); break;
                case 'n': tokens.push_back(read_keyword("null", TokenType::Null)); break;
                case '-': case '0': case '1': case '2': case '3':
                case '4': case '5': case '6': case '7': case '8': case '9':
                    tokens.push_back(read_number());
                    break;
                default:
                    throw std::runtime_error(
                        "词法错误 行" + std::to_string(m_line) +
                        "列" + std::to_string(m_column) + " 意外字符：" + c
                    );
            }
        }
        tokens.emplace_back(TokenType::EndOfInput, "", m_line, m_column);
        return tokens;
    }

private:
    bool is_at_end() const {
        return m_pos >= m_input.size();
    }

    char peek() const {
        if (is_at_end()) return '\0';
        return m_input[m_pos];
    }

    char peek_next() const {
        if (m_pos + 1 >= m_input.size()) return '\0';
        return m_input[m_pos + 1];
    }

    char advance() {
        char c = m_input[m_pos++];
        if (c == '\n') {
            m_line++;
            m_column = 1;
        } else {
            m_column++;
        }
        return c;
    }

    void skip_whitespace_and_comments() {
        while (!is_at_end()) {
            char c = peek();
            if (isspace(static_cast<unsigned char>(c))) {
                advance();
            } else if (c == '/') {
                if (peek_next() == '/') {
                    advance(); advance();
                    while (!is_at_end() && peek() != '\n') advance();
                } else if (peek_next() == '*') {
                    advance(); advance();
                    while (!is_at_end()) {
                        if (peek() == '*' && peek_next() == '/') {
                            advance(); advance();
                            break;
                        }
                        advance();
                    }
                } else {
                    break;
                }
            } else {
                break;
            }
        }
    }

    Token read_keyword(const std::string& keyword, TokenType type) {
        size_t start_line = m_line;
        size_t start_column = m_column;
        for (char ch : keyword) {
            if (advance() != ch) {
                throw std::runtime_error(
                    "词法错误 行" + std::to_string(start_line) +
                    "列" + std::to_string(start_column) + " 关键字错误"
                );
            }
        }
        return Token(type, keyword, start_line, start_column);
    }

    Token read_string() {
        size_t start_line = m_line;
        size_t start_column = m_column;
        advance();
        std::string result;

        while (!is_at_end() && peek() != '"') {
            char c = advance();
            if (c == '\\') {
                if (is_at_end())
                    throw std::runtime_error("字符串未闭合");
                char esc = advance();
                switch (esc) {
                    case '"':  result += '"'; break;
                    case '\\': result += '\\'; break;
                    case '/':  result += '/'; break;
                    case 'b':  result += '\b'; break;
                    case 'f':  result += '\f'; break;
                    case 'n':  result += '\n'; break;
                    case 'r':  result += '\r'; break;
                    case 't':  result += '\t'; break;
                    default: throw std::runtime_error("不支持的转义字符");
                }
            } else {
                result += c;
            }
        }
        if (is_at_end())
            throw std::runtime_error("字符串缺少闭合引号");
        advance();
        return Token(TokenType::String, result, start_line, start_column);
    }

    Token read_number() {
        size_t start_line = m_line;
        size_t start_column = m_column;
        size_t start_pos = m_pos;

        if (peek() == '-') advance();

        if (peek() == '0') {
            advance();
            if (isdigit(static_cast<unsigned char>(peek())))
                throw std::runtime_error("数字不能以0开头");
        } else {
            while (isdigit(static_cast<unsigned char>(peek()))) advance();
        }

        if (peek() == '.') {
            advance();
            if (!isdigit(static_cast<unsigned char>(peek())))
                throw std::runtime_error("小数点后必须有数字");
            while (isdigit(static_cast<unsigned char>(peek()))) advance();
        }

        std::string num_str(m_input.data() + start_pos, m_pos - start_pos);
        return Token(TokenType::Number, num_str, start_line, start_column);
    }
};

关键总结（你现在的最优状态）
 
1. 彻底零深拷贝： Lexer  内部只存指针+长度，不复制任何字符。
2. 只写一个构造：不用  const& 、不用  &&  重载，极简。
3. 原有业务代码一行不用改。
4. 生命周期绝对安全：外部  std::string  生命周期比 Lexer 长，无悬空。
5. 性能比  const& 、值传递+move 都强，还最好懂。
